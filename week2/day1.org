* Is 0 true or false? What about "" and nil?
#+BEGIN_SRC io :results output
 if(0, "true" print, "false" print)
#+END_SRC

#+RESULTS:
: true

#+BEGIN_SRC io :results output
 if("", "true" print, "false" print)
#+END_SRC

#+RESULTS:
: true

#+BEGIN_SRC io :results output
 if(nil, "true" print, "false" print)
#+END_SRC

#+RESULTS:
: false
